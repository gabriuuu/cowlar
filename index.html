⚙️ Decodificador de Mensajes (Payload Formatter)

Pega este código en la consola TTN para traducir los bytes que envía el Heltec V4 a números normales:

function decodeUplink(input) { var b = input.bytes; // Trama 14 bytes · Heltec ESP32-S3 V4 + ADXL345 + GPS L76K var lat = (b[0]|(b[1]<<8)|(b[2]<<16)|(b[3]<<24)) / 1e6; var lng = (b[4]|(b[5]<<8)|(b[6]<<16)|(b[7]<<24)) / 1e6; var pitch = ((b[8]<<8)|b[9]) / 10.0; var roll = ((b[10]<<8)|b[11]) / 10.0; var motion = b[12] / 100.0; var batt_v = b[13] / 50.0; var batt_pct = Math.min(100, Math.max(0, Math.round((batt_v - 3.2) / (4.15 - 3.2) * 100))); var activity = "Reposo"; if (motion > 0.7) activity = "Caminando"; else if (pitch <= -22.0) activity = "Pastando"; else if (pitch > -18 && motion >= 0.08) activity = "Rumia"; return { data: { lat, lng, pitch, roll, motion, batt_v, batt_pct, activity } }; }
1 Entra en console.cloud.thethings.network a la app cowlar-ganado-uets.
2 Entra a Payload formatters › Uplink y pega el código anterior.
